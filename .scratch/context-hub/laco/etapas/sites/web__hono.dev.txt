HonoWeb application framework
Fast, lightweight, built on Web Standards. Support for any JavaScript runtime.

Get Started
View on GitHub
$ npm create hono@latest

import { Hono } from 'hono'

const app = new Hono()
app.get('/', (c) => c.text('Hello Hono!'))

export default app
