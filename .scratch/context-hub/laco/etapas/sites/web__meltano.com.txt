analyze/datasets/closed-won-by-week.yml

version: datasets/v0.2
source: Salesforce
title: Closed-won revenue by week
questions: How much revenue did we close each week this quarter?
description: |-
 Weekly closed-won amount from fct_revenue.

 #revenue #salesforce
metadata: |-
 {
 "name": "fct_revenue",
 "label": "Closed-won revenue",
 "related_table": {
 "columns": [{"name": "week", "label": "Week"}],
 "aggregates": [{"name": "amount", "label": "Revenue"}]
 }
 }
visualisation: '{"chartjs-chart": {"chartType": "bar"}}'
query: |-
 SELECT
 date_trunc('week', close_date) "fct_revenue.week"
 , sum(amount) "fct_revenue.amount"
 FROM fct_revenue
 WHERE close_date >= date_trunc('quarter', current_date)
 GROUP BY 1
 ORDER BY 1
