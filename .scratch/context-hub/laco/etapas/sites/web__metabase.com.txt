Embedded Analytics

Put production-grade analytics into your app without drama

Embed dashboards, visualizations, and self-serve reporting in your
 SaaS app. Multi-tenant, customizable analytics components, with
 plumbing and permissions built in.

import {
 MetabaseProvider,
 InteractiveQuestion,
 defineEmbeddingSdkConfig
} from "@metabase/embedding-sdk-react";

const config = defineEmbeddingSdkConfig({
 metabaseInstanceUrl: "https://metabase.example.com", // Required
 authProviderUri: "https://app.example.com/sso/metabase", // SSO
});

<MetabaseProvider config={config}>
 <InteractiveQuestion questionId={42} />
</MetabaseProvider>
